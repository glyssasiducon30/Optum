				SELECT

					[Date], [Phone Hours]

					SELECT 
						[Date], [Phone Hours]
					FROM #SubQueryPhoneHoursOverallDayWiseMemerServices

		----- Overall Month Wise ------

						''Monthly'' AS Type,''1'' AS Align,base.[MonthNum] AS ''Date'',base.Full_name, base.MSID, base.EMP_ID, base.[Ready Time], base.[Busy Time], base.[Wrap Time], base.[Phone Hours],base.SUPERVISOR,

					SELECT 
						''Daily'' AS Type,''2'' AS Align,cast(base.[Date] AS varchar(30)) AS ''Date'',base.Full_name, base.MSID, base.EMP_ID, base.[Ready Time], base.[Busy Time], base.[Wrap Time], base.[Phone Hours],base.SUPERVISOR,

					FROM #baseraw base
					INNER JOIN #PhoneHoursOverallDayWiseMemerServices PHDaily ON PHDaily.[Date]=base.[Date]
					
				) PHFinalResultSet
