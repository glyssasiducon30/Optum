                charindex(')', Agent_Name) - charindex('(', Agent_Name) - 1) as MSID,

				sum(n_Avg_Engage_Time) as n_Avg_Engage_Time,
				sum(d_Avg_Engage_Time) as d_Avg_Engage_Time,
